        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(0, 12, 22);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(window.devicePixelRatio);
        document.body.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.4);
        scene.add(ambientLight);

        const sunLight = new THREE.PointLight(0xfff5cc, 2, 50);
        scene.add(sunLight);

        const starsGeometry = new THREE.BufferGeometry();
        const starsCount = 1500;
        const starPos = new Float32Array(starsCount * 3);
        for(let i=0; i<starsCount*3; i++) {
            starPos[i] = (Math.random() - 0.5) * 200;
        }
        starsGeometry.setAttribute('position', new THREE.BufferAttribute(starPos, 3));
        const starsMaterial = new THREE.PointsMaterial({ color: 0xffffff, size: 0.5 });
        const starField = new THREE.Points(starsGeometry, starsMaterial);
        scene.add(starField);

        const planets = [];

        marketData.forEach(data => {
            if (data.distance > 0) {
                const ringGeo = new THREE.RingGeometry(data.distance - 0.03, data.distance + 0.03, 64);
                const ringMat = new THREE.MeshBasicMaterial({ color: 0xffffff, side: THREE.DoubleSide, transparent: true, opacity: 0.15 });
                const ring = new THREE.Mesh(ringGeo, ringMat);
                ring.rotation.x = Math.PI / 2;
                scene.add(ring);
            }

            const geo = new THREE.SphereGeometry(data.size, 32, 32);
            let mat;
            if (data.distance === 0) {
                mat = new THREE.MeshBasicMaterial({ color: data.color });
            } else {
                mat = new THREE.MeshStandardMaterial({ color: data.color, roughness: 0.4, metalness: 0.3 });
            }

            const mesh = new THREE.Mesh(geo, mat);
            mesh.userData = { ...data, angle: Math.random() * Math.PI * 2 };
            scene.add(mesh);
            planets.push(mesh);
        });

        const raycaster = new THREE.Raycaster();
        const mouse = new THREE.Vector2();

        window.addEventListener('click', (event) => {
            if (document.getElementById('modal').style.display === 'block') return;

            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObjects(planets);

            if (intersects.length > 0) {
                const selectedData = intersects[0].object.userData;
                openModal(selectedData);
            }
        });

        function openModal(data) {
            const modal = document.getElementById('modal');
            const cat = document.getElementById('modal-category');
            cat.innerText = data.category;
            cat.style.background = '#' + data.color.toString(16);
            cat.style.color = data.color === 0xfacc15 ? '#000' : '#fff';
            
            document.getElementById('modal-title').innerText = data.name;
            document.getElementById('modal-content').innerHTML = data.details;
            modal.style.display = 'block';
        }

        function closeModal() {
            document.getElementById('modal').style.display = 'none';
        }

        function animate() {
            requestAnimationFrame(animate);

            planets.forEach(planet => {
                const data = planet.userData;
                planet.rotation.y += 0.01;

                if (data.distance > 0) {
                    data.angle += data.speed;
                    planet.position.x = Math.cos(data.angle) * data.distance;
                    planet.position.z = Math.sin(data.angle) * data.distance;
                }
            });

            controls.update();
            renderer.render(scene, camera);
        }

        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        animate();
